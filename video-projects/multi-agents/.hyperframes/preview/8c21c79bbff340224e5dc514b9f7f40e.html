14 """Returns every product in the store with its id, name and price."""
15 return {"products": PRODUCTS}

18def add_to_cart(product_id: str, quantity: int, tool_context: ToolContext) -> dict:
19 """Adds a product to the customer's cart (stored in the session state)."""
20 if product_id not in PRODUCTS:
21 return {"error": f"No product found with id {product_id}"}
22
23 cart = tool_context.state.get("cart", {})
24 cart[product_id] = cart.get(product_id, 0) + quantity
25 tool_context.state["cart"] = cart
26 return {"status": "added", "cart": cart}
27

11MODEL = os.getenv("GEMINI_MODEL", "gemini-2.5-flash")
12
13
14def view_cart(tool_context: ToolContext) -> dict:
15 """Returns the items in the cart and the total price."""
16 cart = tool_context.state.get("cart", {})
17 items = []
18 total = 0.0
19 for product_id, quantity in cart.items():
20 product = PRODUCTS[product_id]
21 items.append({"id": product_id, "name": product["name"], "quantity": quantity})
22 total += product["price"] * quantity
23 return {"items": items, "total": round(total, 2)}
24
25
26def remove_from_cart(product_id: str, tool_context: ToolContext) -> dict:
27 """Removes a product from the cart."""
28 cart = tool_context.state.get("cart", {})
29 if product_id not in cart:
30 return {"error": f"{product_id} is not in the cart"}
31
32 del cart[product_id]
33 tool_context.state["cart"] = cart
34 return {"status": "removed", "cart": cart}
35
36
37def confirm_cart(tool_context: ToolContext) -> dict:
38 """Locks in the cart: saves the total and marks the cart as confirmed."""
39 summary = view_cart(tool_context)
40 if not summary["items"]:
41 return {"error": "The cart is empty"}
42
43 tool_context.state["cart_total"] = summary["total"]
44 tool_context.state["cart_confirmed"] = True
45 return {"status": "confirmed", "total": summary["total"]}
46

48cart_agent = Agent(
49 name="cart_agent",
50 model=MODEL,
51 description="Shows the cart, adds or removes items, and confirms the cart.",
52 instruction="""
53 You manage the shopping cart for {user_name?}.
54 - Start by using view_cart to show the items and the total.
55 - Use add_to_cart or remove_from_cart when the customer asks for changes.
56 - If the customer wants to browse more products, transfer to catalog_agent.
57 - When the customer is happy, use confirm_cart, then transfer to order_agent.
58 """,
59 tools=[view_cart, add_to_cart, remove_from_cart, confirm_cart],
60)

14def place_order(tool_context: ToolContext) -> dict:
15 """Places the order for everything in the confirmed cart."""
16 if not tool_context.state.get("cart_confirmed"):
17 return {"error": "Please confirm the cart first"}
18 summary = view_cart(tool_context)
19
20 order_id = "ORD-" + uuid.uuid4().hex[:6].upper()
21 tool_context.state["order_id"] = order_id
22 tool_context.state["order_status"] = "placed"
23 return {
24 "order_id": order_id,
25 "items": summary["items"],
26 "total": summary["total"],
27 "email": tool_context.state.get("user_email"),
28 }
29

31order_agent = Agent(
32 name="order_agent",
33 model=MODEL,
34 description="Confirms the customer's details and places the order.",
35 instruction="""
36 You place the order for {user_name?}.
37 - Use view_cart to show the final items. The total is {cart_total?}.
38 - Confirm the delivery details: email {user_email?}, phone {user_phone?}.
39 - When the customer says yes, use place_order and share the order id.
40 """,
41 tools=[view_cart, place_order],
42)
